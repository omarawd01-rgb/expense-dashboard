function parseData(response) { const rows = response.rawRows || []; totalIncome = parseFloat(response.totalIncome) || 0; allExpenses = []; allIncome = []; // Parse Expenses from Columns A, B, C (Rows 2 down) for (let i = 1; i < rows.length; i++) { const isPaid = rows[i][0] === true || String(rows[i][0]).trim().toUpperCase() === 'TRUE'; const amount = parseFloat(String(rows[i][1] || '0').replace(/[^0-9.-]+/g, "")) || 0; const desc = rows[i][2] || ''; if (amount > 0 && desc.trim() !== '') { allExpenses.push({ rowIndex: i + 1, isPaid, desc, amount }); } } allExpenses.sort((a, b) => b.amount - a.amount); // Parse Main Income items from Column D & E (Rows 3 to 4) if (rows.length >= 5) { // Monthly Salary if (rows[2] && rows[2][3] && rows[2][4]) { allIncome.push({ source: String(rows[2][3]), amount: parseFloat(rows[2][4]) || 0 }); } // House Rental Income if (rows[3] && rows[3][3] && rows[3][4]) { allIncome.push({ source: String(rows[3][3]), amount: parseFloat(rows[3][4]) || 0 }); } } // Parse Extra Income items starting at Row 12 (Index 11) in Column D & E for (let i = 11; i < rows.length; i++) { const source = rows[i] ? rows[i][3] : ''; const amount = rows[i] ? parseFloat(String(rows[i][4] || '0').replace(/[^0-9.-]+/g, "")) || 0 : 0; if (source && source.toString().trim() !== '' && amount > 0) { allIncome.push({ source: String(source), amount }); } } updateMetricsAndCharts(); renderTable(); renderIncomeTable(); }
