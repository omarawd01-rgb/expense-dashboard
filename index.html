function parseData(payload) { const expRows = payload.expenses || payload; const incRows = payload.income || []; // 1. Parse Expenses allExpenses = []; if (expRows && expRows.length > 1) { for (let i = 1; i < expRows.length; i++) { const row = expRows[i]; const isPaid = row[0] === true || String(row[0]).trim().toUpperCase() === 'TRUE'; // Clean numbers (strip out non-numeric chars except dot/minus) const rawAmount = String(row[1] || '0').replace(/[^0-9.-]+/g, ""); const amount = parseFloat(rawAmount) || 0; const desc = row[2] || row[1] || 'Expense Item'; if (amount > 0 || desc !== 'Expense Item') { allExpenses.push({ rowIndex: i + 1, isPaid, desc, amount }); } } } allExpenses.sort((a, b) => b.amount - a.amount); // 2. Parse Income & Compute Total Income Dynamically allIncome = []; totalIncome = 0; if (incRows && incRows.length > 1) { for (let i = 1; i < incRows.length; i++) { const row = incRows[i]; const source = row[0] || 'Income Source'; // Extract numeric value from Col 2 (or Col 1 if single column) let rawAmount = String(row[1] || row[0] || '0').replace(/[^0-9.-]+/g, ""); let amount = parseFloat(rawAmount) || 0; // Skip header row strings if present if (isNaN(amount) || source.toLowerCase() === 'source name' || source.toLowerCase() === 'amount') { continue; } allIncome.push({ rowIndex: i + 1, source, amount }); totalIncome += amount; } } // Fallback: If no income rows were found in 'Income' sheet, populate default salary/rental if (allIncome.length === 0) { allIncome = [ { rowIndex: 2, source: 'Monthly salary (550 USD)', amount: 27000 }, { rowIndex: 3, source: 'House rental income', amount: 11030 } ]; totalIncome = 38030; } updateMetricsAndCharts(); renderTable(); renderIncomeTable(); }
